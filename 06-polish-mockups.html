#include <bits/stdc++.h>
using namespace std;
int main(){
  int n; cin >> n;
  long long ans=0, bal=0;
  map<long long,long long> cnt;
  cnt[0]=1;
  for(int i=0;i<n;i++){
    long long x; cin>>x;
    bal+=(x&1)?1:-1;
    ans+=cnt[bal]++;
  }
  cout<<ans<<"\n";
}
